#include <Arduino.h>
#include <Wire.h>

#include "src/RTC_Module.h"
#include "src/SD_Module.h"

#include "src/Sensors/MAX30102_Module.h"
#include "src/Algo/MAX30109_Processes.h"

#include "src/Sensors/Temperature_Sensor.h"

// ======= Configurações gerais =======
static const int SDA_PIN = 27;
static const int SCL_PIN = 22;

static const int   kSPS     = 100;      // taxa esperada do MAX30102
static const ulong kTickMs  = 1000;     // janela ~1s para consolidar SPO2/BPM

static const char* kUser    = "TESTE";
static const char* kSessionFallback = "YYYYMMDD-00"; // usado só se RTC inválido

// ======= Estado de detecção =======
static bool g_has_max  = false;
static bool g_has_temp = false;

// ======= Controle de janela =======
static unsigned long g_lastTick = 0;
static uint32_t g_samplesThisSec = 0;

// ======= Helpers =======
static void build_session_id(char* out, size_t n)
{
  DateTime t = rtc_getTime();
  // Se o RTC estiver sem hora, cairá num valor “2000..”; tudo bem para teste
  snprintf(out, n, "%04d%02d%02d-00", t.year(), t.month(), t.day());
}

void setup()
{
  Serial.begin(115200);
  delay(100);

  Wire.begin(SDA_PIN, SCL_PIN);

  rtc_init();      // tolera ausência e usa data "2000-01-01" como fallback
  CSV_begin();     // prepara /MBI/CSV/<ANO>/MBI_<YYYYMMDD>.csv

  // --- MAX30102 (SpO2/BPM) ---
  g_has_max = max30102_begin();
  if (g_has_max)
  {
    Serial.println("[MAX] OK");
    ppg_init(kSPS);
    // desliga o gate de dedo para garantir leitura agora; ligue depois se quiser:
    ppg_setFingerThreshold(0);
    max30102_start();
  }
  else
  {
    Serial.println("[MAX] AUSENTE (ou I2C/addr 0x57 falhou)");
  }

  // --- MLX90614 (Temperatura) ---
  init_temperature_sensor();
  // Tentativa rápida de leitura para inferir presença
  float t0 = read_temperature();
  g_has_temp = (!isnan(t0) && t0 > -200.0f && t0 < 120.0f);
  if (g_has_temp) Serial.println("[MLX] OK");
  else            Serial.println("[MLX] AUSENTE");

  g_lastTick = millis();

  Serial.println("[TEST] Rodando. Coloque o dedo no MAX; aguarde ~6–12 s para SpO2/BPM estáveis.");
}

void loop()
{
  // ---------- Ingestão MAX30102 ----------
  if (g_has_max)
  {
    while (max30102_available())
    {
      uint32_t red = 0, ir = 0;
      if (max30102_readRaw(&red, &ir))
      {
        ppg_feedSample(red, ir);
        g_samplesThisSec++;
      }
      else
      {
        break;
      }
    }
  }

  // ---------- Tick ~1s para consolidar SPO2/BPM ----------
  const unsigned long now = millis();
  if (now - g_lastTick >= kTickMs)
  {
    g_lastTick = now;

    // Sessão e time decoram cada linha no CSV
    char session[20];
    build_session_id(session, sizeof(session));

    // Diagnóstico de taxa de amostragem do MAX
    if (g_has_max)
    {
      Serial.print("[DBG] MAX samples/s = ");
      Serial.println((unsigned long)g_samplesThisSec);
    }
    g_samplesThisSec = 0;

    // Fecha janela do PPG (SpO2/BPM)
    if (g_has_max)
    {
      ppg_tick_1s();

      if (ppg_hasSpO2())
      {
        const float spo2 = ppg_getSpO2();
        bool ok = CSV_appendRow(kUser, session, "SPO2", (double)spo2, "%");
        Serial.print("[SPO2] ");
        Serial.print(spo2, 1);
        Serial.print(" %, CSV=");
        Serial.println(ok ? "OK" : "ERRO");
      }
      else
      {
        Serial.println("[SPO2] aguardando janela/condições…");
      }

      if (ppg_hasBPM())
      {
        const float bpm = ppg_getBPM();
        bool ok = CSV_appendRow(kUser, session, "BPM", (double)bpm, "bpm");
        Serial.print("[BPM] ");
        Serial.print(bpm, 1);
        Serial.print(" bpm, CSV=");
        Serial.println(ok ? "OK" : "ERRO");
      }
      else
      {
        Serial.println("[BPM] ainda não detectado…");
      }
    }

    // ---------- Temperatura ----------
    if (g_has_temp)
    {
      float tc = read_temperature(); // retorna -999 em falha
      if (!isnan(tc) && tc > -200.0f && tc < 120.0f)
      {
        bool ok = CSV_appendRow(kUser, session, "TEMP_C", (double)tc, "C");
        Serial.print("[TEMP] ");
        Serial.print(tc, 2);
        Serial.print(" C, CSV=");
        Serial.println(ok ? "OK" : "ERRO");
      }
      else
      {
        Serial.println("[TEMP] leitura inválida");
      }
    }
  }

  delay(2);
}
